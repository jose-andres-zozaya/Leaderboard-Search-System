cmake_minimum_required(VERSION 3.16)
project(LeaderboardSearch LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB SOURCES CONFIGURE_DEPENDS src/*.cpp)

add_executable(leaderboard ${SOURCES})
target_include_directories(leaderboard PRIVATE src)
target_compile_definitions(leaderboard PRIVATE DEFAULT_DATA_PATH="${CMAKE_CURRENT_SOURCE_DIR}/data/players_blitz.csv")

if(WIN32)
    include(FetchContent)
    FetchContent_Declare(pdcurses GIT_REPOSITORY https://github.com/wmcbrine/PDCurses.git GIT_TAG 3.9)
    FetchContent_MakeAvailable(pdcurses)
    file(GLOB PDC_SOURCES ${pdcurses_SOURCE_DIR}/pdcurses/*.c ${pdcurses_SOURCE_DIR}/wincon/*.c)
    add_library(pdcurses STATIC ${PDC_SOURCES})
    target_include_directories(pdcurses PUBLIC ${pdcurses_SOURCE_DIR})
    target_link_libraries(leaderboard PRIVATE pdcurses)
else()
    find_package(Curses REQUIRED)
    target_include_directories(leaderboard PRIVATE ${CURSES_INCLUDE_DIRS})
    target_link_libraries(leaderboard PRIVATE ${CURSES_LIBRARIES})
endif()
